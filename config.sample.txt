# pngshot-sys-screenuploader config
#
# Install to: ux0:data/pngshot-ssu/config.txt
#
# `upload_url` is REQUIRED. It's the endpoint a sys-screenuploader-
# compatible server listens on; pngshot-ssu will send a POST with
# Content-Type: image/png and the raw PNG as the body.
# You can get the upload URL destination id at https://screenuploader.bakatrouble.me/
#
# The special token `{filename}` is expanded to the synthesised
# sys-screenuploader filename  YYYYMMDDhhmmss00-<32×hex>.png  so your
# server can use the same filename parsing as the stock Switch tool.
#
# Use ONLY plaintext http - we use raw sockets

upload_url = http://screenuploader.bakatrouble.me/upload/PUT_DEST_ID_HERE/?filename={filename}

# Set to 0 to disable uploads without removing the plugin.
enabled = 1
